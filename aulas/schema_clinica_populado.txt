-- ------------------------------------------------------------
-- 1. CRIAÇÃO DO BANCO
-- --------------------------------------------biblioteca----------------

CREATE DATABASE IF NOT EXISTS clinica
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE clinica;


-- ============================================================
-- 2. EXCLUSÃO DAS TABELAS
--    (facilita executar o script novamente durante as aulas)
-- ============================================================

SET FOREIGN_KEY_CHECKS = 0;

DROP TABLE IF EXISTS Prescricao;
DROP TABLE IF EXISTS Consulta;
DROP TABLE IF EXISTS Especializacao;
DROP TABLE IF EXISTS Paciente;
DROP TABLE IF EXISTS Especialidade;
DROP TABLE IF EXISTS Medico;

SET FOREIGN_KEY_CHECKS = 1;


-- ============================================================
-- 3. CRIAÇÃO DAS TABELAS
-- ============================================================


-- ------------------------------------------------------------
-- TABELA: MEDICO
-- ------------------------------------------------------------

CREATE TABLE Medico (
    codigo INT NOT NULL AUTO_INCREMENT,
    CRM CHAR(20) NOT NULL,
    nome VARCHAR(50) NOT NULL,
    telefone VARCHAR(20),

    CONSTRAINT pk_medico
        PRIMARY KEY (codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- TABELA: ESPECIALIDADE
-- ------------------------------------------------------------

CREATE TABLE Especialidade (
    codigo INT NOT NULL AUTO_INCREMENT,
    descricao VARCHAR(100) NOT NULL,

    CONSTRAINT pk_especialidade
        PRIMARY KEY (codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- TABELA: ESPECIALIZACAO
-- Relação entre Médico e Especialidade
-- ------------------------------------------------------------

CREATE TABLE Especializacao (
    fkEspecialidade INT NOT NULL,
    fkMedico INT NOT NULL,

    CONSTRAINT pk_especializacao
        PRIMARY KEY (fkEspecialidade, fkMedico),

    CONSTRAINT fk_especializacao_especialidade
        FOREIGN KEY (fkEspecialidade)
        REFERENCES Especialidade(codigo),

    CONSTRAINT fk_especializacao_medico
        FOREIGN KEY (fkMedico)
        REFERENCES Medico(codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- TABELA: PACIENTE
-- ------------------------------------------------------------

CREATE TABLE Paciente (
    codigo INT NOT NULL AUTO_INCREMENT,
    cpf VARCHAR(11) NOT NULL,
    nome VARCHAR(50) NOT NULL,
    telefone VARCHAR(20),
    dt_nascimento DATE,

    CONSTRAINT pk_paciente
        PRIMARY KEY (codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- TABELA: CONSULTA
-- ------------------------------------------------------------

CREATE TABLE Consulta (
    idConsulta INT NOT NULL AUTO_INCREMENT,
    fkPaciente INT NOT NULL,
    fkMedico INT NOT NULL,
    Data DATETIME NOT NULL,
    Diagnostico VARCHAR(255),

    CONSTRAINT pk_consulta
        PRIMARY KEY (idConsulta),

    CONSTRAINT fk_consulta_paciente
        FOREIGN KEY (fkPaciente)
        REFERENCES Paciente(codigo),

    CONSTRAINT fk_consulta_medico
        FOREIGN KEY (fkMedico)
        REFERENCES Medico(codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- TABELA: PRESCRICAO
-- ------------------------------------------------------------

CREATE TABLE Prescricao (
    idPrescricao INT NOT NULL AUTO_INCREMENT,
    nome_medicamento VARCHAR(100) NOT NULL,
    principio_ativo VARCHAR(100),
    dosagem VARCHAR(100),
    fkConsulta INT NOT NULL,

    CONSTRAINT pk_prescricao
        PRIMARY KEY (idPrescricao),

    CONSTRAINT fk_prescricao_consulta
        FOREIGN KEY (fkConsulta)
        REFERENCES Consulta(idConsulta)
) ENGINE=InnoDB;
