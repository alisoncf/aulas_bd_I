-- ============================================================
-- 1. CRIAÇÃO DO BANCO
-- ============================================================

CREATE DATABASE IF NOT EXISTS biblioteca
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE biblioteca;


-- ============================================================
-- 2. EXCLUSÃO DAS TABELAS
-- ============================================================

SET FOREIGN_KEY_CHECKS = 0;

DROP TABLE IF EXISTS emprestimo;
DROP TABLE IF EXISTS autoria;
DROP TABLE IF EXISTS exemplar;
DROP TABLE IF EXISTS membro;
DROP TABLE IF EXISTS livro;
DROP TABLE IF EXISTS autor;
DROP TABLE IF EXISTS editora;
DROP TABLE IF EXISTS tipo_membro;

SET FOREIGN_KEY_CHECKS = 1;


-- ============================================================
-- 3. CRIAÇÃO DAS TABELAS
-- ============================================================


-- ------------------------------------------------------------
-- AUTOR
-- ------------------------------------------------------------

CREATE TABLE autor (
    codigo INT NOT NULL AUTO_INCREMENT,
    nome VARCHAR(60) NOT NULL,
    nacionalidade VARCHAR(60),

    CONSTRAINT pk_autor
        PRIMARY KEY (codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- EDITORA
-- ------------------------------------------------------------

CREATE TABLE editora (
    codEditora INT NOT NULL AUTO_INCREMENT,
    nomeEditora VARCHAR(60) NOT NULL,

    CONSTRAINT pk_editora
        PRIMARY KEY (codEditora)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- TIPO DE MEMBRO
-- ------------------------------------------------------------

CREATE TABLE tipo_membro (
    codigo INT NOT NULL AUTO_INCREMENT,
    descricao VARCHAR(60) NOT NULL,

    CONSTRAINT pk_tipo_membro
        PRIMARY KEY (codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- LIVRO
-- ------------------------------------------------------------

CREATE TABLE livro (
    codigo INT NOT NULL AUTO_INCREMENT,
    ISBN VARCHAR(12) NOT NULL,
    titulo VARCHAR(100) NOT NULL,
    ano INT,
    fkEditora INT NOT NULL,

    CONSTRAINT pk_livro
        PRIMARY KEY (codigo),

    CONSTRAINT uq_livro_isbn
        UNIQUE (ISBN),

    CONSTRAINT fk_livro_editora
        FOREIGN KEY (fkEditora)
        REFERENCES editora(codEditora)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- EXEMPLAR
-- ------------------------------------------------------------

CREATE TABLE exemplar (
    codigo INT NOT NULL AUTO_INCREMENT,
    fkLivro INT NOT NULL,
    disponivel TINYINT(1) NOT NULL DEFAULT 1,
    numTombo VARCHAR(10) NOT NULL,

    CONSTRAINT pk_exemplar
        PRIMARY KEY (codigo),

    CONSTRAINT uq_exemplar_tombo
        UNIQUE (numTombo),

    CONSTRAINT fk_exemplar_livro
        FOREIGN KEY (fkLivro)
        REFERENCES livro(codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- AUTORIA
-- Relação N:N entre Autor e Livro
-- ------------------------------------------------------------

CREATE TABLE autoria (
    fkAutor INT NOT NULL,
    fkLivro INT NOT NULL,
    ordem INT,

    CONSTRAINT pk_autoria
        PRIMARY KEY (fkAutor, fkLivro),

    CONSTRAINT fk_autoria_autor
        FOREIGN KEY (fkAutor)
        REFERENCES autor(codigo),

    CONSTRAINT fk_autoria_livro
        FOREIGN KEY (fkLivro)
        REFERENCES livro(codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- MEMBRO
-- ------------------------------------------------------------

CREATE TABLE membro (
    codigo INT NOT NULL AUTO_INCREMENT,
    matricula VARCHAR(20) NOT NULL,
    fkTipoMembro INT NOT NULL,

    CONSTRAINT pk_membro
        PRIMARY KEY (codigo),

    CONSTRAINT fk_membro_tipo
        FOREIGN KEY (fkTipoMembro)
        REFERENCES tipo_membro(codigo)
) ENGINE=InnoDB;


-- ------------------------------------------------------------
-- EMPRÉSTIMO
-- ------------------------------------------------------------

CREATE TABLE emprestimo (
    numero INT NOT NULL AUTO_INCREMENT,
    fkExemplar INT NOT NULL,
    fkMembro INT NOT NULL,
    data_retirada DATE NOT NULL,
    data_devolucao DATE,

    CONSTRAINT pk_emprestimo
        PRIMARY KEY (numero),

    CONSTRAINT fk_emprestimo_exemplar
        FOREIGN KEY (fkExemplar)
        REFERENCES exemplar(codigo),

    CONSTRAINT fk_emprestimo_membro
        FOREIGN KEY (fkMembro)
        REFERENCES membro(codigo)
) ENGINE=InnoDB;

-- ============================================================
-- 5. CONFERÊNCIA DOS DADOS
-- ============================================================

SELECT 'autor' AS tabela, COUNT(*) AS quantidade
FROM autor

UNION ALL

SELECT 'editora', COUNT(*)
FROM editora

UNION ALL

SELECT 'tipo_membro', COUNT(*)
FROM tipo_membro

UNION ALL

SELECT 'livro', COUNT(*)
FROM livro

UNION ALL

SELECT 'autoria', COUNT(*)
FROM autoria

UNION ALL

SELECT 'exemplar', COUNT(*)
FROM exemplar

UNION ALL

SELECT 'membro', COUNT(*)
FROM membro

UNION ALL

SELECT 'emprestimo', COUNT(*)
FROM emprestimo;
